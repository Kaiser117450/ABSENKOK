```dart
class AdminPolicyRecapDatasetResult {
  final List<AttendancePolicyRecapDay> mergedRows;
  final List<AttendancePolicyRecapDay> strictRows;
  final List<AttendancePolicyRecapDay> fallbackRows;
  bool get isCompatibilityMode;
}

class AdminPolicyRecapDatasetService {
  const AdminPolicyRecapDatasetService({
    LegacyPayrollRecapFallbackService fallbackService = const LegacyPayrollRecapFallbackService(),
  });

  AdminPolicyRecapDatasetResult build({
    required Iterable<Employee> employees,
    required Iterable<AttendancePolicyRecapDay> strictRows,
    required Iterable<AttendanceLog> attendanceLogs,
    required String outletId,
    required String outletName,
    required OutletOperatingMode outletOperatingMode,
    DateTime? now,
  });
}
```

Locked merge rules:
```text
- strict row wins if strict and fallback share the same (employeeId, logicalDate)
- fallback rows are allowed only when attendance-backed logs exist for that key
- fallback rows may produce hadirTanpaJadwal, shortWork, excessBreak, overtime, belumAbsenPulang, or activeIncomplete
- fallback rows may not fabricate late, absence, or schedule-derived cutoff values
```
