**File 1: `sql/phase23_rpc_functions.sql`**

Contains 4 RPC functions using `CREATE OR REPLACE FUNCTION` (additive, safe for production). All functions use `SECURITY DEFINER` with `search_path = public` to bypass RLS (they enforce their own access control internally). Each function takes typed parameters.

```sql
-- ============================================================
-- Phase 23: RPC Functions for Dashboard Aggregation (DASH-05)
-- Safe to run multiple times (CREATE OR REPLACE)
-- ============================================================

-- 1. get_attendance_rates: Returns attendance percentage for an outlet in a date range
-- Counts distinct employees who have at least one 'masuk' log per logical day
-- Returns: { total_employees INT, days_in_range INT, total_present INT, rate NUMERIC }
CREATE OR REPLACE FUNCTION get_attendance_rates(
  p_outlet_id UUID,
  p_start TIMESTAMPTZ,
  p_end TIMESTAMPTZ
)
RETURNS JSON
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_total_employees INT;
  v_days INT;
  v_total_present INT;
BEGIN
  -- Count active (non-archived) employees in this outlet
  SELECT COUNT(*) INTO v_total_employees
  FROM employees
  WHERE outlet_id = p_outlet_id
    AND (archived_at IS NULL);

  -- Count working days in range (exclude Sundays if needed, but for now count all days)
  v_days := GREATEST(1, (p_end::date - p_start::date) + 1);

  -- Count distinct (employee_id, date) pairs with masuk in range
  SELECT COUNT(*) INTO v_total_present
  FROM (
    SELECT DISTINCT employee_id, (scanned_at::date) as scan_date
    FROM attendance_logs
    WHERE scan_outlet_id = p_outlet_id
      AND type = 'masuk'
      AND scanned_at >= p_start
      AND scanned_at < p_end
  ) sub;

  RETURN json_build_object(
    'total_employees', v_total_employees,
    'days_in_range', v_days,
    'total_present', v_total_present,
    'rate', CASE WHEN v_total_employees * v_days > 0
              THEN ROUND((v_total_present::numeric / (v_total_employees * v_days)) * 100, 1)
              ELSE 0 END
  );
END;
$$;

-- 2. get_weekly_trend: Returns daily masuk counts for last N days for an outlet
-- Returns: JSON array of { date TEXT, count INT }
CREATE OR REPLACE FUNCTION get_weekly_trend(
  p_outlet_id UUID,
  p_days INT DEFAULT 7
)
RETURNS JSON
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_result JSON;
BEGIN
  SELECT json_agg(row_to_json(t)) INTO v_result
  FROM (
    SELECT
      d::date::text as date,
      COALESCE(cnt, 0) as count
    FROM generate_series(
      CURRENT_DATE - (p_days - 1),
      CURRENT_DATE,
      '1 day'::interval
    ) d
    LEFT JOIN (
      SELECT scanned_at::date as scan_date, COUNT(DISTINCT employee_id) as cnt
      FROM attendance_logs
      WHERE scan_outlet_id = p_outlet_id
        AND type = 'masuk'
        AND scanned_at >= CURRENT_DATE - (p_days - 1)
      GROUP BY scanned_at::date
    ) logs ON logs.scan_date = d::date
    ORDER BY d
  ) t;

  RETURN COALESCE(v_result, '[]'::json);
END;
$$;

-- 3. get_outlet_comparison: Returns attendance rates across ALL outlets for a date range
-- Admin-only function (caller must be admin)
-- Returns: JSON array of { outlet_id UUID, outlet_name TEXT, rate NUMERIC, total_present INT, total_employees INT }
CREATE OR REPLACE FUNCTION get_outlet_comparison(
  p_start TIMESTAMPTZ,
  p_end TIMESTAMPTZ
)
RETURNS JSON
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_result JSON;
  v_days INT;
BEGIN
  v_days := GREATEST(1, (p_end::date - p_start::date) + 1);

  SELECT json_agg(row_to_json(t)) INTO v_result
  FROM (
    SELECT
      o.id as outlet_id,
      o.name as outlet_name,
      COALESCE(emp.emp_count, 0) as total_employees,
      COALESCE(att.present_count, 0) as total_present,
      CASE WHEN COALESCE(emp.emp_count, 0) * v_days > 0
        THEN ROUND((COALESCE(att.present_count, 0)::numeric / (emp.emp_count * v_days)) * 100, 1)
        ELSE 0 END as rate
    FROM outlets o
    LEFT JOIN (
      SELECT outlet_id, COUNT(*) as emp_count
      FROM employees
      WHERE archived_at IS NULL
      GROUP BY outlet_id
    ) emp ON emp.outlet_id = o.id
    LEFT JOIN (
      SELECT scan_outlet_id, COUNT(DISTINCT (employee_id, scanned_at::date)) as present_count
      FROM attendance_logs
      WHERE type = 'masuk'
        AND scanned_at >= p_start
        AND scanned_at < p_end
      GROUP BY scan_outlet_id
    ) att ON att.scan_outlet_id = o.id
    ORDER BY o.name
  ) t;

  RETURN COALESCE(v_result, '[]'::json);
END;
$$;

-- 4. update_employee_streak: Recalculate and cache streak for one employee
-- Uses noon-rule: logical day runs from noon to noon (12:00 to 11:59 next day)
-- A masuk scan at 22:00 belongs to the logical day starting at that day's noon
-- Returns: JSON { current_streak INT, longest_streak INT }
CREATE OR REPLACE FUNCTION update_employee_streak(
  p_employee_id UUID
)
RETURNS JSON
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_current_streak INT := 0;
  v_longest_streak INT := 0;
  v_prev_logical_day DATE := NULL;
  v_streak INT := 0;
  rec RECORD;
BEGIN
  -- Walk through masuk records, compute logical day using noon rule
  -- Logical day = if hour >= 12 then current_date, else previous_date
  FOR rec IN
    SELECT DISTINCT
      CASE
        WHEN EXTRACT(HOUR FROM scanned_at) >= 12 THEN scanned_at::date
        ELSE (scanned_at - INTERVAL '1 day')::date
      END as logical_day
    FROM attendance_logs
    WHERE employee_id = p_employee_id
      AND type = 'masuk'
    ORDER BY logical_day DESC
  LOOP
    IF v_prev_logical_day IS NULL THEN
      -- First record
      v_streak := 1;
      v_prev_logical_day := rec.logical_day;
    ELSIF v_prev_logical_day - rec.logical_day = 1 THEN
      -- Consecutive day
      v_streak := v_streak + 1;
      v_prev_logical_day := rec.logical_day;
    ELSE
      -- Gap found
      IF v_current_streak = 0 THEN
        v_current_streak := v_streak;
      END IF;
      IF v_streak > v_longest_streak THEN
        v_longest_streak := v_streak;
      END IF;
      v_streak := 1;
      v_prev_logical_day := rec.logical_day;
    END IF;
  END LOOP;

  -- Handle final streak
  IF v_current_streak = 0 THEN
    v_current_streak := v_streak;
  END IF;
  IF v_streak > v_longest_streak THEN
    v_longest_streak := v_streak;
  END IF;

  -- Upsert into employee_streaks cache table
  INSERT INTO employee_streaks (employee_id, current_streak, longest_streak, last_masuk_date, updated_at)
  VALUES (p_employee_id, v_current_streak, v_longest_streak,
    (SELECT MAX(CASE WHEN EXTRACT(HOUR FROM scanned_at) >= 12 THEN scanned_at::date ELSE (scanned_at - INTERVAL '1 day')::date END)
     FROM attendance_logs WHERE employee_id = p_employee_id AND type = 'masuk'),
    NOW()
  )
  ON CONFLICT (employee_id) DO UPDATE SET
    current_streak = EXCLUDED.current_streak,
    longest_streak = EXCLUDED.longest_streak,
    last_masuk_date = EXCLUDED.last_masuk_date,
    updated_at = NOW();

  RETURN json_build_object(
    'current_streak', v_current_streak,
    'longest_streak', v_longest_streak
  );
END;
$$;
```

```sql
-- ============================================================
-- Phase 23: Employee Streaks Cache Table (GAME-01)
-- Additive migration — safe for production
-- ============================================================

-- Create table (IF NOT EXISTS = safe to run multiple times)
CREATE TABLE IF NOT EXISTS employee_streaks (
  employee_id UUID PRIMARY KEY REFERENCES employees(id) ON DELETE CASCADE,
  current_streak INT NOT NULL DEFAULT 0,
  longest_streak INT NOT NULL DEFAULT 0,
  last_masuk_date DATE,
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Enable RLS
ALTER TABLE employee_streaks ENABLE ROW LEVEL SECURITY;

-- RLS: Admin and Kepala Gerai can read streaks for employees in their outlets
DROP POLICY IF EXISTS "streaks_select_admin" ON employee_streaks;
CREATE POLICY "streaks_select_admin" ON employee_streaks
  FOR SELECT
  USING (
    -- Admin can see all
    (auth.jwt() -> 'app_metadata' ->> 'app_role') = 'admin'
    OR
    -- Kepala Gerai can see streaks for employees in their managed outlet
    (
      (auth.jwt() -> 'app_metadata' ->> 'app_role') = 'kepala_gerai'
      AND employee_id IN (
        SELECT id FROM employees
        WHERE outlet_id = (auth.jwt() -> 'app_metadata' ->> 'managed_outlet_id')::uuid
      )
    )
  );

-- RLS: Only RPC functions (SECURITY DEFINER) can insert/update
-- No direct client insert/update policy needed since update_employee_streak runs as SECURITY DEFINER
DROP POLICY IF EXISTS "streaks_insert_service" ON employee_streaks;
CREATE POLICY "streaks_insert_service" ON employee_streaks
  FOR INSERT
  WITH CHECK (false); -- only SECURITY DEFINER functions bypass this

DROP POLICY IF EXISTS "streaks_update_service" ON employee_streaks;
CREATE POLICY "streaks_update_service" ON employee_streaks
  FOR UPDATE
  USING (false); -- only SECURITY DEFINER functions bypass this

-- Grant usage
GRANT SELECT ON employee_streaks TO authenticated;
```
