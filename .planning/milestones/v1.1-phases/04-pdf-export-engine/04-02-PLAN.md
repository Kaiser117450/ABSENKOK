    ```dart
    Future<void> _exportPdf() async {
      // Guard: must have data
      if (!_hasLoaded || _dailyRows.isEmpty) {
        if (mounted) {
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(content: Text('Muat data terlebih dahulu')),
          );
        }
        return;
      }

      setState(() => _exportingPdf = true);
      try {
        final summaries = _computeDailySummaries();

        // Resolve outlet name — "Semua Outlet" if no specific outlet selected
        String outletName = 'Semua Outlet';
        if (_selectedOutletId != null) {
          final outlet = _outlets.firstWhere(
            (o) => o.id == _selectedOutletId,
            orElse: () => _outlets.first,
          );
          outletName = outlet.name;
        }

        await PdfReportService.generateAttendanceReport(
          summaries: summaries,
          startDate: _startDate,
          endDate: _endDate,
          outletName: outletName,
        );
      } catch (e) {
        if (mounted) {
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text('Gagal generate PDF: $e')),
          );
        }
      } finally {
        if (mounted) setState(() => _exportingPdf = false);
      }
    }
    ```

    CRITICAL: `setState(() => _exportingPdf = true)` MUST be called BEFORE the `await`
    so the loading spinner appears immediately. Do NOT call setState after the await starts.
  </action>
